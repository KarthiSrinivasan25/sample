class Accounts{
  constructor(Name, Dob, AccNo){
    this.Name = Name,
    this.Dob = Dob,
    this.AccNo = AccNo
  }
  display(){
    console.log("---------Account Details--------")
  console.log(this.Name)
  console.log(this.Dob)
  console.log(this.AccNo)
  }
  
}

class Savings extends Accounts{
  restictions(){
    console.log("--------Savings Accounts Restictions---------")
    console.log("Minimum Balance Maintain 500rs")
    console.log("Per-Day Withdraw Limits 5Times")
  }
}

class Current extends Accounts{
  restictions(){
    console.log("--------Current Accounts Restictions---------")
    console.log("No Minimum Balance")
    console.log("Unlimited Withdraw")
  }
}

let a1 = new Savings("Karthi", "25/08/200", 123456789)
a1.display()
a1.restictions()

let a2 = new Current("Swetha", "01/08/2001", 987654321)
a2.display()
a2.restictions()
